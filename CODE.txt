
class Book:
    def __init__(self, book_id, title, author, copies):
        self.book_id = book_id
        self.title = title
        self.author = author
        self.copies = copies


class Library:
    def __init__(self):
        self.books = {}
        self.reservations = {}
        self.waiting_list = {}
        self.sequence = 0

    def add_book(self):
        try:
            book_id = int(input("Enter Book ID: "))

            if book_id in self.books:
                print("Book already exists!")
                return

            title = input("Enter Book Title: ")
            author = input("Enter Author: ")
            copies = int(input("Enter Number of Copies: "))

            if copies < 0:
                print("Invalid number of copies!")
                return

            self.books[book_id] = Book(
                book_id, title, author, copies
            )
            self.reservations[book_id] = []
            self.waiting_list[book_id] = []

            print("Book added successfully!")

        except ValueError:
            print("Enter valid details!")

    def display_books(self):
        if not self.books:
            print("No books available!")
            return

        print("\n--- AVAILABLE BOOKS ---")

        for book_id in sorted(self.books):
            book = self.books[book_id]
            print("ID:", book.book_id)
            print("Title:", book.title)
            print("Author:", book.author)
            print("Available Copies:", book.copies)
            print("----------------------")

    def search_book(self):
        keyword = input("Enter title or author: ").lower()
        found = False

        for book in self.books.values():
            if (keyword in book.title.lower() or
                    keyword in book.author.lower()):
                print("Book ID:", book.book_id)
                print("Title:", book.title)
                print("Author:", book.author)
                print("Available Copies:", book.copies)
                found = True

        if not found:
            print("Book not found!")

    def reserve_book(self):
        try:
            book_id = int(input("Enter Book ID: "))
            student = input("Enter Student Name: ").strip()
            priority = int(input("Priority (1-High, 2-Medium, 3-Low): "))

            if book_id not in self.books:
                print("Invalid Book ID!")
                return

            if not student or priority not in [1, 2, 3]:
                print("Invalid details!")
                return

            if student in self.reservations[book_id]:
                print("Already reserved!")
                return

            if any(r[2] == student for r in self.waiting_list[book_id]):
                print("Already in waiting list!")
                return

            book = self.books[book_id]

            if book.copies > 0 and not self.waiting_list[book_id]:
                book.copies -= 1
                self.reservations[book_id].append(student)
                print("Book reserved successfully!")
            else:
                self.sequence += 1
                request = [priority, self.sequence, student]
                self.waiting_list[book_id].append(request)

                print("Book unavailable!")
                print("Student added to waiting list.")

            self.save_data()

        except ValueError:
            print("Enter valid input!")

    def process_waiting_list(self, book_id):
        queue = self.waiting_list[book_id]

        if self.books[book_id].copies > 0 and queue:
            queue.sort(key=lambda x: (x[0], x[1]))

            request = queue.pop(0)
            student = request[2]

            self.books[book_id].copies -= 1
            self.reservations[book_id].append(student)

            print("\n*** BOOK AVAILABLE NOTIFICATION ***")
            print("Hello", student + "!")
            print("Your reserved book",
                  self.books[book_id].title,
                  "is now available.")

    def return_book(self):
        try:
            book_id = int(input("Enter Book ID: "))
            student = input("Enter Student Name: ").strip()

            if book_id not in self.books:
                print("Invalid Book ID!")
                return

            if student not in self.reservations[book_id]:
                print("No active reservation found!")
                return

            self.reservations[book_id].remove(student)
            self.books[book_id].copies += 1

            print("Book returned successfully!")

            self.process_waiting_list(book_id)
            self.save_data()

        except ValueError:
            print("Invalid input!")

    def display_waiting_list(self):
        try:
            book_id = int(input("Enter Book ID: "))

            if book_id not in self.books:
                print("Invalid Book ID!")
                return

            queue = sorted(
                self.waiting_list[book_id],
                key=lambda x: (x[0], x[1])
            )

            if not queue:
                print("Waiting list is empty!")
                return

            print("\n--- WAITING LIST ---")

            for request in queue:
                priority = request[0]
                student = request[2]

                levels = {
                    1: "High",
                    2: "Medium",
                    3: "Low"
                }

                print("Student:", student,
                      "| Priority:", levels[priority])

        except ValueError:
            print("Invalid input!")

    def display_reservations(self):
        print("\n--- ACTIVE RESERVATIONS ---")
        found = False

        for book_id, students in self.reservations.items():
            for student in students:
                print("Book:", self.books[book_id].title,
                      "| Student:", student)
                found = True

        if not found:
            print("No active reservations!")

    def save_data(self):
        try:
            with open("library_data.txt", "w") as file:
                for book in self.books.values():
                    file.write(
                        str(book.book_id) + "|" +
                        book.title + "|" +
                        book.author + "|" +
                        str(book.copies) + "\n"
                    )
            print("Book data saved successfully!")
        except OSError:
            print("File storage unavailable!")


def main():
    library = Library()

    while True:
        print("\n===== SMART LIBRARY SYSTEM =====")
        print("1. Add Book")
        print("2. Display Books")
        print("3. Search Book")
        print("4. Reserve Book")
        print("5. Return Book")
        print("6. Display Waiting List")
        print("7. Display Reservations")
        print("8. Exit")

        choice = input("Enter your choice: ")

        if choice == "1":
            library.add_book()
        elif choice == "2":
            library.display_books()
        elif choice == "3":
            library.search_book()
        elif choice == "4":
            library.reserve_book()
        elif choice == "5":
            library.return_book()
        elif choice == "6":
            library.display_waiting_list()
        elif choice == "7":
            library.display_reservations()
        elif choice == "8":
            library.save_data()
            print("Thank you for using Smart Library!")
            break
        else:
            print("Invalid choice!")


if __name__ == "__main__":
    main()
